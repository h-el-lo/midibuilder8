// Validate analog readings are in expected range
bool validateAnalogRead(int reading, int expectedMin, int expectedMax) {
  if (reading < 0 || reading > 1023) {
    // ADC returned invalid value
    return false;
  }
  // Optional: check if value is in calibrated range
  if (reading < expectedMin - 50 || reading > expectedMax + 50) {
    // Sensor may be disconnected or failing
    return false;
  }
  return true;
}

#define ERROR_LED_PIN 13

enum ErrorCode {
  ERR_NONE = 0,
  ERR_KEY_STUCK = 1,
  ERR_INVALID_TIMING = 2,
  ERR_ADC_FAULT = 3,
  ERR_MIDI_OVERFLOW = 4
};

uint8_t errorCode = ERR_NONE;
uint16_t errorCount[5] = {0};  // Count each error type

void logError(ErrorCode code) {
  errorCode = code;
  errorCount[code]++;
  
  // Flash LED in pattern
  for (int i = 0; i < code; i++) {
    digitalWrite(ERROR_LED_PIN, HIGH);
    delay(100);
    digitalWrite(ERROR_LED_PIN, LOW);
    delay(100);
  }
  
  // Also log to serial for debugging
  Serial.print("Error ");
  Serial.print(code);
  Serial.print(": count=");
  Serial.println(errorCount[code]);
}

// If a key appears stuck, automatically send note-off after timeout
const unsigned long KEY_STUCK_TIMEOUT = 5000;  // 5 seconds

void checkStuckKeys() {
  static unsigned long lastCheck = 0;
  
  if (millis() - lastCheck > 1000) {  // Check every second
    for (uint8_t x = 0; x < ROW_NUM; x++) {
      for (uint8_t y = 0; y < COL_NUM; y++) {
        if (not_ready[x][y]) {
          if (millis() - timer[1][x][y] > KEY_STUCK_TIMEOUT) {
            // Force note off
            note = nums[x][y] + transpose;
            noteOff(0, note, 0);
            not_ready[x][y] = 0;
            logError(ERR_KEY_STUCK);
          }
        }
      }
    }
    lastCheck = millis();
  }
}
